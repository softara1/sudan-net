<script type="text/javascript">
    $(document).ready(function () {
        // 1. معالجة أيقونة المشاركة
        $('.posts-icon').each(function () {
            if (!$(this).find('.participate-check').is(':empty')) {
                $(this).find('.participate-check').html('<i class="material-symbols-outlined" title="' + $(this).find('.participate-check').children('img').attr('title') + '">reply</i>');
            }
        });

        // 2. معالجة أيقونة الموضوع القديمة
        $('.posts-description').each(function () {
            var background_image = '';
            if($(this).attr('style')) {
                background_image = $(this).attr('style').replace(/background-image: url\(|\);/g, '');
                $(this).children('h3').prepend('<img class="topic-icon" src="' + background_image + '" />');
                $(this).attr('style', '');
            }
        });

        // 3. معالجة الترقيم الاحترافي الجديد
        $('.msr-mini-pagination').each(function() {
            var $paginationContainer = $(this);
            $paginationContainer.contents().filter(function() { return this.nodeType === 3; }).remove();
            $paginationContainer.find('span:not(.page-sep)').each(function() {
                if($(this).find('a, strong').length === 0) { $(this).remove(); }
            });
            if ($paginationContainer.children().length > 0) { $paginationContainer.css('display', 'inline-flex'); }
        });

        // 4. معالجة الصور الفارغة أو المكسورة لإظهار الأيقونة البديلة في المنتصف
        $('.col-cover .avatarcp img').each(function() {
            var imgSrc = $(this).attr('src');
            // التحقق من الرابط الفارغ أو رابط الصورة الافتراضية لأحلى منتدى
            if (!imgSrc || imgSrc === '' || imgSrc.indexOf('https://2img.net/i/fa/empty.gif') !== -1 || imgSrc === 'undefined') {
                $(this).closest('.col-cover').addClass('empty-cover');
                $(this).hide(); // إخفاء عنصر الصورة الفارغة
            }
        }).on('error', function() {
            // إذا كانت الصورة مكسورة أو لم يتم تحميلها
            $(this).closest('.col-cover').addClass('empty-cover');
            $(this).hide();
        });
    });
</script>
