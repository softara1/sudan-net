    // ===== Reveal on Scroll =====
    var msrRevealObserver = new IntersectionObserver(function(entries) {
        entries.forEach(function(entry) {
            if (entry.isIntersecting) {
                entry.target.classList.add('msr-visible');
            }
        });
    }, { threshold: 0.1, rootMargin: '0px 0px -60px 0px' });

    document.querySelectorAll('.msr-reveal').forEach(function(el) {
        msrRevealObserver.observe(el);
    });

    // ===== Toast =====
    function msrShowToast(msg) {
        var toast = document.getElementById('msrToast');
        var toastMsg = document.getElementById('msrToastMsg');
        toastMsg.textContent = msg;
        toast.classList.add('msr-show');
        setTimeout(function() { toast.classList.remove('msr-show'); }, 3000);
    }
</script>

<script type="text/javascript">
 $(function() {
    // تحديد الألوان
    var colorFirstWord = "#0F172A"; // لون الكلمة الأولى (داكن)
    var colorRestWords = "#2563EB"; // لون الكلمة الثانية فما فوق (أزرق)

    // إضافة تنسيقات CSS
    $('head').append('<style>' +
        '#logo-text, #logo-text h1, #logo-text a, #msr-site-name { visibility: visible !important; opacity: 1 !important; text-decoration: none !important; }' +
        '.colored-part-1 { color: ' + colorFirstWord + ' !important; text-decoration: none !important; border-bottom: none !important; background: none !important; }' +
        '.colored-part-2 { color: ' + colorRestWords + ' !important; text-decoration: none !important; border-bottom: none !important; background: none !important; }' +
        '.colored-part-1::after, .colored-part-2::after { content: none !important; display: none !important; }' +
    '</style>');

    function paintMyTitle(selector) {
        var $container = $(selector);
        if ($container.length === 0) return;

        var $target = $container.find('h1');
        if ($target.length === 0) $target = $container;

        if ($target.find('.colored-part-1').length > 0) return;

        var text = $target.text().trim();

        if (text.length > 0) {
            var parts = text.split(" ");
            if (parts.length > 1) {
                var first = parts.shift();
                var rest = parts.join(" ");
                $target.html(
                    '<span class="colored-part-1">' + first + '</span> ' +
                    '<span class="colored-part-2">' + rest + '</span>'
                );
            } else {
                $target.html('<span class="colored-part-1">' + text + '</span>');
            }
        }
    }

    // تطبيق الدالة على كلا الموقعين
    function applyPaint() {
        paintMyTitle('#logo-text');       // الهيدر العلوي
        paintMyTitle('#msr-site-name');   // بانر الترحيب
    }

    applyPaint();
    setInterval(applyPaint, 500);
});
</script>
