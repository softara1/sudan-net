        createButtons() {
            this.container = document.createElement('div');
            this.container.className = 'scroll-buttons-container-3d';
            // استخدام أيقونات Font Awesome المتوافقة مع القالب
            this.upButton = this.createButton('up', '<i class="fas fa-chevron-up"></i>');
            this.downButton = this.createButton('down', '<i class="fas fa-chevron-down"></i>');
            this.container.appendChild(this.upButton);
            this.container.appendChild(this.downButton);
            document.body.appendChild(this.container);
            setTimeout(() => {
                this.container.classList.add('visible');
            }, 500);
        }
        createButton(type, iconHtml) {
            const btn = document.createElement('button');
            btn.className = `scroll-btn-3d ${type}-btn`;
            const icon = document.createElement('span');
            icon.className = 'btn-icon';
            icon.innerHTML = iconHtml;
            btn.appendChild(icon);
            const percentageContainer = document.createElement('div');
            percentageContainer.className = 'percentage-container';
            const percentageCircle = document.createElement('div');
            percentageCircle.className = 'percentage-circle';
            const percentageText = document.createElement('span');
            percentageText.className = 'percentage-text';
            percentageText.textContent = '0%';
            percentageCircle.appendChild(percentageText);
            percentageContainer.appendChild(percentageCircle);
            btn.appendChild(percentageContainer);
            if (type === 'up') {
                this.upPercentageText = percentageText;
                this.upCircle = percentageCircle;
            } else {
                this.downPercentageText = percentageText;
                this.downCircle = percentageCircle;
            }
            btn.addEventListener('click', (e) => this.handleClick(e, btn, type));
            return btn;
        }
        handleClick(event, btn, type) {
            this.createWaterRipple(event, btn);
            if (type === 'up') {
                window.scrollTo({ top: 0, behavior: 'smooth' });
            } else {
                window.scrollTo({ top: document.documentElement.scrollHeight, behavior: 'smooth' });
            }
        }
        createWaterRipple(event, btn) {
            const rect = btn.getBoundingClientRect();
            const ripple = document.createElement('span');
            ripple.className = 'water-ripple';
            const x = event.clientX - rect.left;
            const y = event.clientY - rect.top;
            ripple.style.left = `${x}px`;
            ripple.style.top = `${y}px`;
            ripple.style.transform = 'translate(-50%, -50%)';
            btn.appendChild(ripple);
            setTimeout(() => ripple.remove(), 800);
        }
        updatePercentageColor(percentage) {
            // تغيير لون الدائرة ليتناسب مع تدرج القالب: سماوي ← أزرق ← بنفسجي
            let color;
            if (percentage < 33) {
                color = '#0EA5E9'; // Sky Blue
            } else if (percentage < 66) {
                color = '#2563EB'; // Primary Blue
            } else {
                color = '#8B5CF6'; // Violet
            }
            if (this.upCircle) this.upCircle.style.background = color;
            if (this.downCircle) this.downCircle.style.background = color;
            if (percentage > 80) {
                this.upCircle.style.animation = 'percentagePulse 1s infinite';
                this.downCircle.style.animation = 'percentagePulse 1s infinite';
            } else {
                this.upCircle.style.animation = 'percentagePulse 3s infinite';
                this.downCircle.style.animation = 'percentagePulse 3s infinite';
            }
        }
        attachEvents() {
            window.addEventListener('scroll', () => this.handleScroll());
            window.addEventListener('resize', () => this.handleScroll());
        }
        handleScroll() {
            const scrollY = window.scrollY;
            const windowHeight = window.innerHeight;
            const documentHeight = document.documentElement.scrollHeight;
            this.scrollPercentage = Math.round((scrollY / (documentHeight - windowHeight)) * 100) || 0;
            if (this.upPercentageText) this.upPercentageText.textContent = `${this.scrollPercentage}%`;
            if (this.downPercentageText) this.downPercentageText.textContent = `${this.scrollPercentage}%`;
            this.updatePercentageColor(this.scrollPercentage);
            if (scrollY < 100) {
                this.upButton.classList.add('disabled');
            } else {
                this.upButton.classList.remove('disabled');
            }
            if (scrollY + windowHeight >= documentHeight - 100) {
                this.downButton.classList.add('disabled');
            } else {
                this.downButton.classList.remove('disabled');
            }
        }
        setPosition(bottom, left) {
            if (bottom !== undefined) {
                this.config.bottomOffset = bottom;
                this.container.style.bottom = `${bottom}px`;
            }
            if (left !== undefined) {
                this.config.sideOffset = left;
                this.container.style.left = `${left}px`;
            }
        }
        updateConfig(newConfig) {
            this.config = { ...this.config, ...newConfig };
            if (newConfig.bottomOffset) this.container.style.bottom = `${newConfig.bottomOffset}px`;
            if (newConfig.sideOffset) this.container.style.left = `${newConfig.sideOffset}px`;
        }
        hide() { this.container.classList.remove('visible'); }
        show() { this.container.classList.add('visible'); }
        destroy() { this.container.remove(); }
    }
    // تهيئة الأزرار عند تحميل الصفحة
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', () => {
            window.scrollButtons3D = new ScrollButtons3D({
                position: 'left',
                bottomOffset: 100,
                sideOffset: 30,
                buttonSize: 65
            });
        });
    } else {
        window.scrollButtons3D = new ScrollButtons3D({
            position: 'left',
            bottomOffset: 100,
            sideOffset: 30,
            buttonSize: 65
        });
    }
})();
